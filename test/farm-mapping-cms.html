  $('btnSubmit').addEventListener('click', async () => {
    // Validate all required steps before submit
    for(let i=0; i { if(val !== undefined && $(id)) $(id).value = val ?? ""; };

    set('fullName', f.fullName);
    set('gender', f.gender);
    set('dob', f.dob);
    set('phone', f.phone);
    set('nationality', f.nationality);
    set('nationalId', f.nationalId);

    set('county', f.county);
    set('district', f.district);
    set('community', f.community);
    set('inspectorName', f.inspectorName);

    set('lat', f.lat);
    set('lng', f.lng);
    $('gpsTime').textContent = f.gpsTime || "Time: —";
    $('gpsStatus').textContent = (f.lat && f.lng) ? "GPS: Loaded" : "GPS: Not captured";

    set('directions', f.directions);

    set('farmName', f.farmName);
    set('crop', f.crop);
    set('ownership', f.ownership);
    set('regStatus', f.regStatus);
    set('farmSizeManual', f.farmSizeManual);
    set('farmUnitManual', f.farmUnitManual);
    set('farmNotes', f.farmNotes);

    $('consent').checked = !!f.consent;

    // Restore GPS marker
    if(f.lat && f.lng){
      const la = Number(f.lat), lo = Number(f.lng);
      if(isFinite(la) && isFinite(lo)){
        const ll = [la, lo];
        if(!gpsMarker) gpsMarker = L.marker(ll).addTo(map);
        else gpsMarker.setLatLng(ll);
      }
    }

    // Restore polygon
    resetPolygon();
    if(d.boundary && d.boundary.geometry){
      const layer = L.geoJSON(d.boundary, { style:{ weight:2, color:"#0b7a33" } }).getLayers()[0];
      currentPolygonLayer = layer;
      drawnItems.addLayer(layer);
      updatePolygonUI();
    }

    // Restore done steps
    done.clear?.();
    (Array.isArray(d.done) ? d.done : []).forEach(x => done.add(x));

    setAutosaveState("Draft: Loaded");
    toast("Loaded", "Draft loaded. Please reattach files before submitting.");
    updateValidationUI();
    currentStep = 0;
    showStep(currentStep);
  }

  $('btnSaveDraftTop').addEventListener('click', saveDraft);
  $('btnLoadDraftTop').addEventListener('click', loadDraft);

  $('btnSaveDraftTop').title = "Save draft locally (no files saved)";
  $('btnLoadDraftTop').title = "Load draft saved on this computer";

  $('btnClearAll').addEventListener('click', () => {
    if(!confirm("Clear all fields in this form?")) return;

    // Clear text/select/textarea
    document.querySelectorAll('input, select, textarea').forEach(el => {
      if(el.type === "file") el.value = "";
      else if(el.type === "checkbox") el.checked = false;
      else el.value = "";
    });

    // Reset GPS + marker
    $('gpsStatus').textContent = "GPS: Not captured";
    $('gpsTime').textContent = "Time: —";
    if(gpsMarker){ map.removeLayer(gpsMarker); gpsMarker = null; }

    // Reset polygon
    resetPolygon();

    // Reset QR
    $('farmerId').value = "";
    $('qrBox').innerHTML = `QR will appear after submit.`;
    qrTextValue = "";
    qrInstance = null;

    // Reset done + step
    done.clear?.();
    currentStep = 0;
    setAutosaveState("Draft: Not saved");
    toast("Cleared", "Form reset.");
    showStep(currentStep);
  });

  // Also allow Save/Load via sidebar (same buttons are in topbar only)
  $('btnPrev').addEventListener('click', prev);
  $('btnNext').addEventListener('click', next);

  // -----------------------------
  // Init
  // -----------------------------
  // Quick top labels
  $('topTitle').textContent = "New Farmer Registration";
  $('topSub').textContent = "Professional CMS-style wizard for field mapping, ID + QR generation.";

  setAutosaveState(getDraft() ? "Draft: Available" : "Draft: Not saved");
  renderStepper();
  showStep(currentStep);
  updateValidationUI();
